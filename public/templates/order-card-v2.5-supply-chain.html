路由代码 · onClick 决策树
前端 store 层注入
// OrderCard onClick 路由 — 一处定，全站用
function getCardClickDest(order, user) {
  // 1. Sanlyn 内部任何角色 → 详情页（全权）
  if (user.tenant === 'sanlyn') return `/admin/orders/${order.id}`;

  // 2. 客户 (买方) → 客户门户详情
  if (user.role === 'customer' && order.customer_id === user.company_id)
    return `/portal/orders/${order.id}`;

  // 3. 供应链 (工厂/司机/报关行) → 协同卡（做事）
  if (['factory', 'trucker', 'broker'].includes(user.role)
      && order.collaborators.includes(user.id))
    return `/collab/${order.collab_code}?as=${user.role}`;

  // 4. 仅持有文件分享 token → 文件包页（无登录）
  if (user.tokens?.fileShare?.[order.id])
    return `/files/${user.tokens.fileShare[order.id]}`;

  // 5. 其他 → 拒绝（fail-closed）
  throw new AccessDenied('No access to this order');
}
